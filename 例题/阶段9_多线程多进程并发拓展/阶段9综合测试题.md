### 编程题参考答案

#### 题目11参考代码
```python
from concurrent.futures import ThreadPoolExecutor
import time

def call_ai_api(prompt):
    time.sleep(0.5)
    return f"AI回复：{prompt}"

prompts = [
    "解释什么是机器学习",
    "Python和Java的区别",
    "如何学习AI开发",
    "推荐一些AI学习资源",
    "什么是大语言模型",
    "如何优化Python代码",
    "AI的未来发展方向",
    "什么是深度学习"
]

# 单线程
start = time.time()
for prompt in prompts:
    call_ai_api(prompt)
print(f"单线程耗时：{time.time()-start:.2f}秒")

# 多线程
start = time.time()
with ThreadPoolExecutor(max_workers=8) as executor:
    results = list(executor.map(call_ai_api, prompts))
print(f"多线程耗时：{time.time()-start:.2f}秒")

# 输出结果
for result in results:
    print(result)
```

#### 题目12参考代码
```python
from concurrent.futures import ThreadPoolExecutor, TimeoutError
import time

def process_task(task_id, duration):
    time.sleep(duration)
    return f"{task_id}：完成"

tasks = [
    ("任务A", 0.5),
    ("任务B", 1.5),
    ("任务C", 2.5),
    ("任务D", 0.8)
]

with ThreadPoolExecutor(max_workers=3) as executor:
    futures = {
        executor.submit(process_task, task_id, duration): task_id
        for task_id, duration in tasks
    }

    for future, task_id in futures.items():
        try:
            result = future.result(timeout=2)
            print(result)
        except TimeoutError:
            print(f"{task_id}：超时")
```
