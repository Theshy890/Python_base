### 编程题参考答案

#### 题目11参考代码
```python
class AIModel:
    def __init__(self, model_name, api_key):
        self.model_name = model_name
        self.__api_key = api_key
        self.temperature = 0.7

    def generate(self, prompt):
        return f"[{self.model_name}] 正在生成：{prompt}"

    def set_temperature(self, temp):
        if 0 <= temp <= 1:
            self.temperature = temp
            print(f"温度已设置为：{temp}")
        else:
            print("温度必须在0-1之间")

    def get_api_key_masked(self):
        return self.__api_key[:4] + "****" + self.__api_key[-4:]

# 测试
model = AIModel("GPT-4", "sk-abcdefghijklmnop")
print(model.model_name)
print(model.get_api_key_masked())
model.set_temperature(0.9)
print(model.generate("你好"))
```

#### 题目12参考代码
```python
class BaseAI:
    def __init__(self, api_key):
        self.api_key = api_key

    def chat_completion(self, prompt):
        pass

class OpenAIModel(BaseAI):
    def chat_completion(self, prompt):
        return f"[OpenAI] 回复：{prompt}"

class ClaudeModel(BaseAI):
    def chat_completion(self, prompt):
        return f"[Claude] 回复：{prompt}"

class AIServiceManager:
    def __init__(self):
        self.services = {}

    def add_service(self, name, service):
        self.services[name] = service

    def call_service(self, name, prompt):
        if name in self.services:
            return self.services[name].chat_completion(prompt)
        else:
            return f"服务 {name} 不存在"

# 测试
manager = AIServiceManager()
manager.add_service("openai", OpenAIModel("sk-123456"))
manager.add_service("claude", ClaudeModel("sk-abcdef"))

result1 = manager.call_service("openai", "你好")
result2 = manager.call_service("claude", "你好")
print(result1)
print(result2)
```
