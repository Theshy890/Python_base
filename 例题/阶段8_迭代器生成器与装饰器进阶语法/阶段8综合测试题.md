### 编程题参考答案

#### 题目11参考代码
```python
import time
import functools

def batch_generator(data, batch_size):
    for i in range(0, len(data), batch_size):
        yield data[i:i+batch_size]

def timer(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"函数 {func.__name__} 执行时间：{end-start:.4f}秒")
        return result
    return wrapper

def retry(max_attempts=3):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise e
                    print(f"第{attempt+1}次失败，重试中...")
        return wrapper
    return decorator

# 测试
data = list(range(100))

@timer
def process_batch(batch):
    return sum(batch)

for batch in batch_generator(data, 20):
    result = process_batch(batch)
    print(f"批次结果：{result}")
```

#### 题目12参考代码
```python
import time
import functools

def retry_on_failure(max_retries=3, delay=1):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_retries - 1:
                        raise e
                    print(f"第{attempt+1}次失败，{delay}秒后重试...")
                    time.sleep(delay)
        return wrapper
    return decorator

def memoize(func):
    cache = {}
    @functools.wraps(func)
    def wrapper(*args):
        if args not in cache:
            cache[args] = func(*args)
            print(f"缓存未命中，执行函数")
        else:
            print(f"缓存命中，直接返回")
        return cache[args]
    return wrapper

@retry_on_failure(max_retries=3, delay=1)
@memoize
def call_ai_api(prompt):
    import random
    if random.random() > 0.5:
        raise Exception("网络超时")
    return f"AI回复：{prompt}"

# 测试
print(call_ai_api("你好"))
print(call_ai_api("你好"))
print(call_ai_api("什么是AI"))
```
